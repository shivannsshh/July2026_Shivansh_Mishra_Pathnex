Jenkinsfile With Git Integration and Testing

pipeline{
    agent any
    stages{
        stage('Clone Repository'){
            steps {
                git "repo link"
            }
        }
        stage('Build') {
            steps {
                echo "Building project from Git repository"
            }
        }
        stage('Test') {
            steps {
                echo "Running unit tests"
            }
        }
        stage('Deploy') {
            steps {
                echo "Deploying Application"
            }
        }
    
    }
}

# Ansible  Create User and Set Permissions

- name: Create a new user
  hosts: all
  become: yes
  tasks:
     - name: Create user
       user:
         name: pathnex
         state: present
         shell: /bin/bash
     - name: Set permission for the user
       file:
          path: /home/pathnex
          owner: pathnex
          group: pathnex
          mode: '0755'
