# Playbook starts from here
- name: Install Nginx on Pathnex server   # Kind of Heading name of the playbook (just for readability)

  hosts: all                              # It runs this playbook on all servers mentioned in the inventory file

  become: yes                             # Execute commands with root (sudo) privileges

  tasks:                                  # List of tasks Ansible will perform

    - name: Install Nginx                 # Task name (displayed while execution)

      yum:                                # Use the YUM package manager (Amazon Linux, CentOS, RHEL)

        name: nginx                       # Package that needs to be installed

        state: present                    # Ensure nginx is installed.
                                          # If already installed, Ansible does nothing.
