#Ansible - Install & Enable Nginx

- name: Install and start Nginx on Pathnex  #only for readability
  hosts: all                #it runs this playbook on all serveres mention in the inventory 
  become: yes               # it execute tasks with root powers

  tasks:                    #list of tasks ansible will perform
    - name: Install nginx   #Task1 Install the nginx package
      yum:
        name: nginx         #package name to install
        state: present      #ensure nginx is installed if already installed, ansible does nothing

    - name: Enable nginx    #Task2 Start and enable the nginx service
      service:              #manage linux services
        name: nginx         #service name
        state: started      #it start the service if it isnt already running
        enabled: yes        # enable nginx to start automatically after every reboot
