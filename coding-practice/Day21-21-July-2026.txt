#Multi-Environment Pipeline


# Jenkinns Pipeline -Deploy to Multiple Environments

#u will learn how to define multiple environemt deployment in one pipeline

pipeline {
   agent any
   environment {
    INSTITUTE_NAME = "Pathnex"
    }

    stages {
        stage('Deploy to Dev') {
            steps {
                sh 'echo "Deploying $INSTITUTE_NAME to Dev environment"'
            }
        }

        stage ('Deploy to staging'){
            steps{
                input message: "Approve Staging Deployment?"
                sh 'echo "Deployment $INSTITUTE_NAME to staging environment"'
            }   
        }
        stage('Deploy to Prod'){
            steps {
                input message: "Approve Prod Deployment?"
                sh 'echo "Deployment $INSTITUTE_NAME to Prod environment"'
            }

            }
    }



}
